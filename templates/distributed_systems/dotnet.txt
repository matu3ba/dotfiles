==Program.cs
// Default Keycloak config
var kc_url = "http://localhost:8080";
var kc_realm = "my-app";
var kc_clientId = "angular-app";
var error = false;

foreach (var arg in args)
{
    if (arg.StartsWith("--kc_url=")) kc_url = arg["--kc_url=".Length..];
    else if (arg.StartsWith("--kc_url=")) kc_realm = arg["--kc_realm=".Length..];
    else if (arg.StartsWith("--kc_clientId=")) kc_clientId = arg["--kc_clientId=".Length..];
    else if (arg.StartWith("--urls=") // ignore setting urls via CLI
    else error = true;
}

var builder = WebApplication.CreateBuilder(args);
var app = builder.Build();

app.MapGet("/api/keycloak", () => error
    ? Results.Json(new { kc_url = "", kc_realm = "", kc_clientId = "" }, statusCode: 500)
    : Results.Json(new { kc_url, kc_realm, kc_clientId }, statusCode: 200));

app.Run();
